275  }
276

278  {

292  Processor[ik]->SetFancyNames(GetFromManager<bool>(Settings["FancyNames"], true, __FILE__, __LINE__));
293  Processor[ik]->Initialise();
294 
295  if(Settings["BurnInSteps"]) {
296  Processor[ik]->SetStepCut(Settings["BurnInSteps"].as<int>());
297  }else {
298  Processor[ik]->SetStepCut(static_cast<int>(Processor[ik]->GetnSteps()/5));
299  }
300 
301  if(Settings["MaxEntries"]) {
302  Processor[ik]->SetEntries(Get<int>(Settings["MaxEntries"], __FILE__, __LINE__));
303  }
304  if(Settings["NBins"]) {
305  Processor[ik]->SetNBins(Get<int>(Settings["NBins"], __FILE__, __LINE__));
306  }
307  }
308 
309  Processor[0]->MakePostfit(this->GetCustomBinning(Settings));
310  Processor[0]->DrawPostfit();
311  // Get edges from first histogram to ensure all params use same binning
312  std::map<std::string, std::pair<double, double>> ParamEdges;
313  for(int i = 0; i < Processor[0]->GetNParams(); ++i) {
314  // Get the histogram for the i-th parameter
315  TH1D* hist = Processor[0]->GetHpost(i);
316  if (!hist) {
317  MACH3LOG_DEBUG("Histogram for parameter {} is null.", i);
318  continue;
319  }
320 
321  // Get the parameter name (title of the histogram)
322  std::string paramName = hist->GetTitle();
323 
324  // Get the axis limits (edges)
325  TAxis* axis = hist->GetXaxis();
326  double xmin = axis->GetXmin();
327  double xmax = axis->GetXmax();
328 
329  MACH3LOG_DEBUG("Adding bin edges for {} equal to {:.4f}, {:.4f}",paramName, xmin, xmax);
330  // Insert into the map
331  ParamEdges[paramName] = std::make_pair(xmin, xmax);
332  }
333 
334  //KS: Multithreading here is very tempting but there are some issues with root that need to be resovled :(
335  for (int ik = 1; ik < nFiles; ik++)
336  {
337  // Make the postfit
338  Processor[ik]->MakePostfit(ParamEdges);
339  Processor[ik]->DrawPostfit();
340  }
341 
342  // Open a TCanvas to write the posterior onto
343  auto Posterior = std::make_unique<TCanvas>("PosteriorMulti", "PosteriorMulti", 0, 0, 1024, 1024);
344  gStyle->SetOptStat(0);
345  gStyle->SetOptTitle(0);
346  Posterior->SetGrid();
347  Posterior->SetBottomMargin(0.1f);
348  Posterior->SetTopMargin(0.05f);
349  Posterior->SetRightMargin(0.03f);
350  Posterior->SetLeftMargin(0.15f);
351 
352  // First filename: keep path, just remove ".root"
353  // Would be nice to specify outpath in a later update

355  std::string base = (pos == std::string::npos) ? FileNames[0] : FileNames[0].substr(0, pos);

393

413

439  hpd[ik]->SetLineColor(hpost[ik]->GetLineColor());
440  hpd[ik]->SetLineWidth(2);
441  hpd[ik]->SetLineStyle(kSolid);

443

445  double maximum = 0;

447  for (int ik = 0; ik < nFiles; ik++) hpost[ik]->SetMaximum(1.3*maximum);
448 
449  hpost[0]->Draw("hist");
450  for (int ik = 1; ik < nFiles; ik++) hpost[ik]->Draw("hist same");
451  Asimov->Draw("same");
452  for (int ik = 0; ik < nFiles; ik++) hpd[ik]->Draw("same");
453  leg->Draw("same");

455  Posterior->Print(canvasname);
456  }//End loop over parameters
457 
458  // Finally draw the parameter plot onto the PDF
459  // Close the .pdf file with all the posteriors
460  Posterior->cd();
461  Posterior->Clear();
462 
463  if(GetFromManager<bool>(Settings["PerformKStest"], true, __FILE__, __LINE__)) this->KolmogorovSmirnovTest(Processor, Posterior, canvasname);
464 
465  // Close the pdf file
466  MACH3LOG_INFO("Closing pdf {}", canvasname);
467  canvasname+="]";
468  Posterior->Print(canvasname);
469  }

599  void ProcessMCMCModule::DiagnoseCovarianceMatrix(MCMCProcessor* Processor, const std::string& inputFile)
600  {
601  //Turn of plots from Processor
602  Processor->SetPrintToPDF(false);
603  // Open a TCanvas to write the posterior onto
604  auto Canvas = std::make_unique<TCanvas>("Canvas", "Canvas", 0, 0, 1024, 1024);
605  Canvas->SetGrid();
606  gStyle->SetOptStat(0);
607  gStyle->SetOptTitle(0);
608  Canvas->SetTickx();
609  Canvas->SetTicky();
610  Canvas->SetBottomMargin(0.1f);
611  Canvas->SetTopMargin(0.05f);
612  Canvas->SetRightMargin(0.15f);
613  Canvas->SetLeftMargin(0.10f);
614 
615  //KS: Fancy colours
616  const int NRGBs = 10;
617  TColor::InitializeColors();
618  Double_t stops[NRGBs] = { 0.00, 0.10, 0.25, 0.35, 0.50, 0.60, 0.65, 0.75, 0.90, 1.00 };
619  Double_t red[NRGBs] = { 0.50, 1.00, 1.00, 0.25, 0.00, 0.10, 0.50, 1.00, 0.75, 0.55 };
620  Double_t green[NRGBs] = { 0.00, 0.25, 1.00, 0.25, 0.00, 0.60, 0.90, 1.00, 0.75, 0.75 };
621  Double_t blue[NRGBs] = { 0.00, 0.25, 1.00, 1.00, 0.50, 0.60, 0.90, 1.00, 0.05, 0.05 };
622  TColor::CreateGradientColorTable(NRGBs, stops, red, green, blue, 255);
623  gStyle->SetNumberContours(255);
624 
625  std::string OutName = inputFile;
626  OutName = OutName.substr(0, OutName.find(".root"));
627  Canvas->Print(Form("Correlation_%s.pdf[", OutName.c_str()), "pdf");
628  Canvas->Print(Form("Covariance_%s.pdf[", OutName.c_str()), "pdf");
629 
630  YAML::Node card_yaml = M3OpenConfig(config.c_str());
631  YAML::Node Settings = card_yaml["ProcessMCMC"];
632 
633  const int entries = int(Processor->GetnSteps());
634  const int NIntervals = GetFromManager<int>(Settings["NIntervals"], 5, __FILE__, __LINE__);
635  const int IntervalsSize = entries/NIntervals;
636  //We start with burn from 0 (no burn in at all)
637  int BurnIn = 0;
638  MACH3LOG_INFO("Diagnosing matrices with entries={}, NIntervals={} and IntervalsSize={}", entries, NIntervals, IntervalsSize);
639 
640  TMatrixDSym *Covariance = nullptr;
641  TMatrixDSym *Correlation = nullptr;
642 
643  TH2D *CovariancePreviousHist = nullptr;
644  TH2D *CorrelationPreviousHist = nullptr;
645 
646  TH2D *CovarianceHist = nullptr;
647  TH2D *CorrelationHist = nullptr;
648 
649  //KS: Get first covariances, we need two for comparison...
650  Processor->SetStepCut(BurnIn);
651  Processor->GetCovariance(Covariance, Correlation);
652 
653  CovariancePreviousHist = this->TMatrixIntoTH2D(Covariance, "Covariance");
654  CorrelationPreviousHist = this->TMatrixIntoTH2D(Correlation, "Correlation");
655 
656  delete Covariance;
657  Covariance = nullptr;
658  delete Correlation;
659  Correlation = nullptr;
660 
661  //KS: Loop over all desired cuts
662  for(int k = 1; k < NIntervals; ++k)
663  {
664  BurnIn = k*IntervalsSize;
665  Processor->SetStepCut(BurnIn);
666  Processor->GetCovariance(Covariance, Correlation);
667  Processor->Reset2DPosteriors();
668 
669  CovarianceHist = this->TMatrixIntoTH2D(Covariance, "Covariance");
670  CorrelationHist = this->TMatrixIntoTH2D(Correlation, "Correlation");
671 
672  TH2D *CovarianceDiff = static_cast<TH2D*>(CovarianceHist->Clone("Covariance_Ratio"));
673  TH2D *CorrelationDiff = static_cast<TH2D*>(CorrelationHist->Clone("Correlation_Ratio"));
674 
675  //KS: Bit messy but quite often covariance is 0 is divided by 0 is problematic so
676  #ifdef MULTITHREAD
677  #pragma omp parallel for
678  #endif
679  for (int j = 1; j < CovarianceDiff->GetXaxis()->GetNbins()+1; ++j)
680  {
681  for (int i = 1; i < CovarianceDiff->GetYaxis()->GetNbins()+1; ++i)
682  {
683  if( std::fabs (CovarianceDiff->GetBinContent(j, i)) < 1.e-5 && std::fabs (CovariancePreviousHist->GetBinContent(j, i)) < 1.e-5)
684  {
685  CovarianceDiff->SetBinContent(j, i, M3::_BAD_DOUBLE_);
686  CovariancePreviousHist->SetBinContent(j, i, M3::_BAD_DOUBLE_);
687  }
688  if( std::fabs (CorrelationDiff->GetBinContent(j, i)) < 1.e-5 && std::fabs (CorrelationPreviousHist->GetBinContent(j, i)) < 1.e-5)
689  {
690  CorrelationDiff->SetBinContent(j, i, M3::_BAD_DOUBLE_);
691  CorrelationPreviousHist->SetBinContent(j, i, M3::_BAD_DOUBLE_);
692  }
693  }
694  }
695  //Divide matrices
696  CovarianceDiff->Divide(CovariancePreviousHist);
697  CorrelationDiff->Divide(CorrelationPreviousHist);
698 
699  //Now it is time for fancy names etc.
700  for (int j = 0; j < CovarianceDiff->GetXaxis()->GetNbins(); ++j)
701  {
702  TString Title = "";
703  double Prior = 1.0;
704  double PriorError = 1.0;
705 
706  Processor->GetNthParameter(j, Prior, PriorError, Title);
707 
708  CovarianceDiff->GetXaxis()->SetBinLabel(j+1, Title);
709  CovarianceDiff->GetYaxis()->SetBinLabel(j+1, Title);
710  CorrelationDiff->GetXaxis()->SetBinLabel(j+1, Title);
711  CorrelationDiff->GetYaxis()->SetBinLabel(j+1, Title);
712  }
713  CovarianceDiff->GetXaxis()->SetLabelSize(0.015f);
714  CovarianceDiff->GetYaxis()->SetLabelSize(0.015f);
715  CorrelationDiff->GetXaxis()->SetLabelSize(0.015f);
716  CorrelationDiff->GetYaxis()->SetLabelSize(0.015f);
717 
718  std::stringstream ss;
719  ss << "BCut_";
720  ss << BurnIn;
721  ss << "/";
722  ss << "BCut_";
723  ss << (k-1)*IntervalsSize;
724  std::string str = ss.str();
725 
726  TString Title = "Cov " + str;
727  CovarianceDiff->GetZaxis()->SetTitle( Title );
728  Title = "Corr " + str;
729  CorrelationDiff->GetZaxis()->SetTitle(Title);
730 
731  CovarianceDiff->SetMinimum(-2);
732  CovarianceDiff->SetMaximum(2);
733  CorrelationDiff->SetMinimum(-2);
734  CorrelationDiff->SetMaximum(2);
735 
736  Canvas->cd();
737  CovarianceDiff->Draw("colz");
738  Canvas->Print(Form("Covariance_%s.pdf", OutName.c_str()), "pdf");
739 
740  CorrelationDiff->Draw("colz");
741  Canvas->Print(Form("Correlation_%s.pdf", OutName.c_str()), "pdf");
742 
743  //KS: Current hist become previous as we need it for further comparison
744  delete CovariancePreviousHist;
745  CovariancePreviousHist = static_cast<TH2D*>(CovarianceHist->Clone());
746  delete CorrelationPreviousHist;
747  CorrelationPreviousHist = static_cast<TH2D*>(CorrelationHist->Clone());
748 
749  delete CovarianceHist;
750  CovarianceHist = nullptr;
751  delete CorrelationHist;
752  CorrelationHist = nullptr;
753 
754  delete CovarianceDiff;
755  delete CorrelationDiff;
756  delete Covariance;
757  Covariance = nullptr;
758  delete Correlation;
759  Correlation = nullptr;
760  }
761  Canvas->cd();
762  Canvas->Print(Form("Covariance_%s.pdf]", OutName.c_str()), "pdf");
763  Canvas->Print(Form("Correlation_%s.pdf]", OutName.c_str()), "pdf");
764 
765  Processor->SetPrintToPDF(true);
766  if(Covariance != nullptr) delete Covariance;
767  if(Correlation != nullptr) delete Correlation;
768  if(CovariancePreviousHist != nullptr) delete CovariancePreviousHist;
769  if(CorrelationPreviousHist != nullptr) delete CorrelationPreviousHist;
770  if(CovarianceHist != nullptr) delete CovarianceHist;
771  if(CorrelationHist != nullptr) delete CorrelationHist;
772  }

805  void ProcessMCMCModule::KolmogorovSmirnovTest(const std::vector<std::unique_ptr<MCMCProcessor>>& Processor,
806  const std::unique_ptr<TCanvas>& Posterior,
807  const TString& canvasname)
808  {
809  constexpr Color_t CumulativeColor[] = {kBlue-1, kRed, kGreen+2};
810  constexpr Style_t CumulativeStyle[] = {kSolid, kDashed, kDotted};
811 
812  for(int i = 0; i < Processor[0]->GetNParams(); ++i)
813  {
814  // This holds the posterior density
815  std::vector<std::unique_ptr<TH1D>> hpost(nFiles);
816  std::vector<std::unique_ptr<TH1D>> CumulativeDistribution(nFiles);
817 
818  TString Title;
819  double Prior = 1.0;
820  double PriorError = 1.0;
821 
822  Processor[0]->GetNthParameter(i, Prior, PriorError, Title);
823  bool Skip = false;
824  for (int ik = 0 ; ik < nFiles; ik++)
825  {
826  int Index = 0;
827  if(ik == 0 ) Index = i;
828  else
829  {
830  // KS: If somehow this chain doesn't given params we skip it
831  Index = Processor[ik]->GetParamIndexFromName(hpost[0]->GetTitle());
832  if(Index == M3::_BAD_INT_)
833  {
834  Skip = true;
835  break;
836  }
837  }
838  hpost[ik] = M3::Clone(Processor[ik]->GetHpost(Index));
839  CumulativeDistribution[ik] = M3::Clone(Processor[ik]->GetHpost(Index));
840  CumulativeDistribution[ik]->Fill(0., 0.);
841  CumulativeDistribution[ik]->Reset();
842  CumulativeDistribution[ik]->SetMaximum(1.);
843  TString TempTitle = Title+" Kolmogorov Smirnov";
844  CumulativeDistribution[ik]->SetTitle(TempTitle);
845 
846  TempTitle = Title+" Value";
847  CumulativeDistribution[ik]->GetXaxis()->SetTitle(TempTitle);
848  CumulativeDistribution[ik]->GetYaxis()->SetTitle("Cumulative Probability");
849 
850  CumulativeDistribution[ik]->SetLineWidth(2);
851  CumulativeDistribution[ik]->SetLineColor(CumulativeColor[ik]);
852  CumulativeDistribution[ik]->SetLineStyle(CumulativeStyle[ik]);
853  }
854 
855  // Don't plot if this is a fixed histogram (i.e. the peak is the whole integral)
856  if(hpost[0]->GetMaximum() == hpost[0]->Integral()*1.5 || Skip) {
857  continue;
858  }
859 
860  for (int ik = 0 ; ik < nFiles; ik++)
861  {
862  const int NumberOfBins = hpost[ik]->GetXaxis()->GetNbins();
863  double Cumulative = 0;
864  const double Integral = hpost[ik]->Integral();
865  for (int j = 1; j < NumberOfBins+1; ++j)
866  {
867  Cumulative += hpost[ik]->GetBinContent(j)/Integral;
868  CumulativeDistribution[ik]->SetBinContent(j, Cumulative);
869  }
870  //KS: Set overflow to 1 just in case
871  CumulativeDistribution[ik]->SetBinContent(NumberOfBins+1, 1.);
872  }
873 
874  std::vector<int> TestStatBin(nFiles, 0);
875  std::vector<double> TestStatD(nFiles, -999);
876  std::vector<std::unique_ptr<TLine>> LineD(nFiles);
877  //Find KS statistic
878  for (int ik = 1 ; ik < nFiles; ik++)
879  {
880  const int NumberOfBins = CumulativeDistribution[0]->GetXaxis()->GetNbins();
881  for (int j = 1; j < NumberOfBins+1; ++j)
882  {
883  const double BinValue = CumulativeDistribution[0]->GetBinCenter(j);
884  const int BinNumber = CumulativeDistribution[ik]->FindBin(BinValue);
885  //KS: Calculate D statistic for this bin, only save it if it's bigger than previously found value
886  double TempDstat = std::fabs(CumulativeDistribution[0]->GetBinContent(j) - CumulativeDistribution[ik]->GetBinContent(BinNumber));
887  if(TempDstat > TestStatD[ik])
888  {
889  TestStatD[ik] = TempDstat;
890  TestStatBin[ik] = j;
891  }
892  }
893  }
894 
895  for (int ik = 0 ; ik < nFiles; ik++)
896  {
897  LineD[ik] = std::make_unique<TLine>(CumulativeDistribution[0]->GetBinCenter(TestStatBin[ik]), 0, CumulativeDistribution[0]->GetBinCenter(TestStatBin[ik]), CumulativeDistribution[0]->GetBinContent(TestStatBin[ik]));
898  LineD[ik]->SetLineColor(CumulativeColor[ik]);
899  LineD[ik]->SetLineWidth(2.0);
900  }
901  CumulativeDistribution[0]->Draw();
902  for (int ik = 0 ; ik < nFiles; ik++)
903  CumulativeDistribution[ik]->Draw("SAME");
904 
905  auto leg = std::make_unique<TLegend>(0.15, 0.7, 0.5, 0.90);
906  leg->SetTextSize(0.04f);
907  for (int ik = 0; ik < nFiles; ik++)
908  leg->AddEntry(CumulativeDistribution[ik].get(), TitleNames[ik].c_str(), "l");
909  for (int ik = 1; ik < nFiles; ik++)
910  leg->AddEntry(LineD[ik].get(), Form("#Delta D = %.4f", TestStatD[ik]), "l");
911 
912  leg->SetLineColor(0);
913  leg->SetLineStyle(0);
914  leg->SetFillColor(0);
915  leg->SetFillStyle(0);
916  leg->Draw("SAME");
917 
918  for (int ik = 1; ik < nFiles; ik++)
919  LineD[ik]->Draw("sam");
920 
921  Posterior->cd();
922  Posterior->Print(canvasname);
923  } //End loop over parameter
924  }
