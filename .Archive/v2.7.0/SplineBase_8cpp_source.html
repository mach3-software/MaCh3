56 // Takes down the number of binary searches from 1.2M to 17, ha!
57 // For ROOT version see root/hist/hist/src/TSpline3.cxx TSpline3::FindX(double)
58 void SplineBase::FindSplineSegment() {
59 // *************************
60  // Loop over the splines
61  //KS: Tried multithreading here with 48 splines and it is faster with one thread, maybe in future multithreading will be worth revisiting
62  for (M3::int_t i = 0; i < nParams; ++i)
63  {
64  const M3::int_t nPoints = SplineInfoArray[i].nPts;
65  const std::vector<M3::float_t>& xArray = SplineInfoArray[i].xPts;
66 
67  // Get the variation for this reconfigure for the ith parameter
68  const float xvar = float(*SplineInfoArray[i].splineParsPointer);
69  ParamValues[i] = xvar;
70 
71  // EM: if we have a parameter that has no response for any event (i.e. all splines have just one knot), then skip it and avoid a seg fault here
72  // In principle, such parameters shouldn't really be included in the first place, but with new det syst splines this
73  // could happen if say you were just running on one FHC run, then all RHC parameters would be flat and the code below would break.
74  if(xArray.size() == 0) continue;
75 
76  // The segment we're interested in (klow in ROOT code)
77  M3::int_t segment = 0;
78  M3::int_t kHigh = nPoints-1;
79  //KS: We expect new segment is very close to previous
80  const M3::int_t PreviousSegment = SplineInfoArray[i].CurrSegment;
81 
82  // If the variation is below the lowest saved spline point
83  if (xvar <= xArray[0]) {
84  segment = 0;
85  // If the variation is above the highest saved spline point
86  } else if (xvar >= xArray[nPoints-1]) {
87  //CW: Yes, the -2 is indeed correct, see TSpline.cxx:814 and //see: https://savannah.cern.ch/bugs/?71651
88  segment = kHigh;
89  //KS: It is quite probable the new segment is same as in previous step so try to avoid binary search, first we have to check if it is in bounds to avoid seg fault
90  } else if( xArray[PreviousSegment+1] > xvar && xvar >= xArray[PreviousSegment] ) {
91  segment = PreviousSegment;
92  // If the variation is between the maximum and minimum, perform a binary search
93  } else {
94  // The top point we've got
95  M3::int_t kHalf = 0;
96  // While there is still a difference in the points (we haven't yet found the segment)
97  // This is a binary search, incrementing segment and decrementing kHalf until we've found the segment
98  while (kHigh - segment > 1) {
99  // Increment the half-step
100  kHalf = M3::int_t((segment + kHigh)/2);
101  // If our variation is above the kHalf, set the segment to kHalf
102  if (xvar > xArray[kHalf]) {
103  segment = kHalf;
104  // Else move kHigh down
105  } else {
106  kHigh = kHalf;
107  }
108  } // End the while: we've now done our binary search
109  } // End the else: we've now found our point
110 
111  if (segment >= nPoints-1 && nPoints > 1) segment = nPoints-2;
112 
113  // CW: This way we avoid doing 1.2M+ binary searches on the GPU
114  // and literally just multiply lots of numbers together on the GPU without any algorithm
115  // Update the values and which segment it belongs to
116  SplineInfoArray[i].CurrSegment = segment;
117  SplineSegments[i] = short(SplineInfoArray[i].CurrSegment);
118 
119  #ifdef MACH3_DEBUG
120  CheckSegmentValidity(i, segment, xvar, SplineInfoArray[i].xPts);
121  #endif
122  } //end loop over params
123 }
