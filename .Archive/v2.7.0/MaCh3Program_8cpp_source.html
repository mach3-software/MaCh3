187  std::vector<std::string> MaCh3Program::expand_plugin_path(const std::string& path) const {
188  std::vector<std::string> result;
189 
190  fs::path p(path);
191 
192  if (!fs::exists(p)) {
193  std::cerr << "Plugin path does not exist: " << path << "\n";
194  return result;
195  }
196 
197  if (fs::is_regular_file(p)) {
198  // Single .so file
199  result.push_back(path);
200  }
201  else if (fs::is_directory(p)) {
202  // Load all .so files in directory
203  for (auto& entry : fs::directory_iterator(p)) {
204  if (entry.is_regular_file() && entry.path().extension() == ".so") {
205  result.push_back(entry.path().string());
206  }
207  }
208  }
209  else {
210  std::cerr << "Invalid plugin path (not file or directory): " << path << "\n";
211  }
212 
213  return result;
214  }
