116  const std::size_t width,
117  const Alignment alignment)
118  {
119  if (value.size() >= width){
120  return std::string(value);
121  }
122  const std::size_t padding = width - value.size();
123 
124  switch (alignment) {
125  case Alignment::Left:

128  case Alignment::Right:

131  case Alignment::Center: {
132  const std::size_t left = padding / 2;
133  const std::size_t right = padding - left;
134 
135  return std::string(left, ' ') + std::string(value) + std::string(right, ' ');
136  }
137  }
