110 // von Mises window function (circular analog of Gaussian)
111 double vonMisesWindow(double x, double center, double kappa) {
112  // von Mises PDF: exp(kappa * cos(x - center)) / (2*pi*I0(kappa))
113  // For numerical stability, compute in log space when possible
114  double I0_kappa;
115  if (kappa > 700) {
116  // Use asymptotic approximation for large kappa to avoid overflow
117  // log(I0(kappa)) ≈ kappa - 0.5*log(2*pi*kappa)
118  double log_I0 = kappa - 0.5 * log(2 * TMath::Pi() * kappa);
119  return exp(kappa * cos(x - center) - log_I0 - log(2 * TMath::Pi()));
120  } else {
121  I0_kappa = TMath::BesselI0(kappa);
122  return exp(kappa * cos(x - center)) / (2 * TMath::Pi() * I0_kappa);
123  }
124 }

126 // Generalised Gaussian window allows harsher constraints on the tails of the distribution, better for controlling windows in highly disfavoured regions
127 double generalisedGaussian2(double x, double mean, double width) {
128  constexpr int n = 2; // this controls the tightness of the gaussian fixed at 2
129  // for now due to normalisation
130  const double normFactor = 1 / ((M3::UmbrellaGaussianNormFactor) * 2 * std::sqrt(2) * width); // the normalisation is a little ugly (uses gamma functions),
131  // im just going to hardcode them for now
132  double likelihood = normFactor * std::exp(-std::pow((std::pow(x - mean, 2) / (2 * std::pow(width, 2))), n));
133  return likelihood;
134 }

137 double GetMulticanonicalWeightGenGaussian(double deltacp, double mean, double width) {
138  // implementation of the generalised gaussian as a bias function
139  // for now with a fixed n = 2 for simplicity
140 
141  double g0 = generalisedGaussian2(deltacp, mean, width);
142  double g1 = generalisedGaussian2(deltacp, mean - 2 * TMath::Pi(),width); // these two repeats are required for wrapping the gaussian around -+pi
143  double g2 = generalisedGaussian2(deltacp, mean + 2 * TMath::Pi(), width);
144  double multicanonicalBeta = 1.0;
145  return (g0 + g1 + g2) * (multicanonicalBeta);
146 }

273 // Main function to calculate the F matrix from a given set of samples plus weights
274 std::vector<std::vector<double>> calcFmatrix(std::vector<double> &z_current,
275  const std::vector<WindowConfig> &windows,
276  const std::vector<std::vector<double>> &samples,
277  const std::vector<std::vector<std::vector<double>>> &window_cache) {
278  int n_windows = static_cast<int>(windows.size());
279  std::vector<std::vector<double>> F(n_windows, std::vector<double>(n_windows, 0.0));
280 
281  std::vector<double> z_inv = z_current; // make a copy to avoid modifying the original z_current
282  for (size_t i = 0; i < z_current.size(); i++) {
283  if (z_current[i] > 0) {
284  z_inv[i] = 1.0 / z_current[i];
285  } else {
286  z_inv[i] = 0.0; // Handle zero or negative z values gracefully
287  if (debug_mode) {
288  MACH3LOG_WARN("Warning: z_current[{}] is non-positive ({}). Setting its inverse to 0 in F matrix calculation.", i, z_current[i]);
289  }
290  }
291  }
292  #ifdef MULTITHREAD
293  #pragma omp parallel for schedule(dynamic)
294  #endif
295  for (int i = 0; i < n_windows; i++) {
296  std::vector<double> denominator_cache(samples[i].size(), 0.0);
297  for (size_t s = 0; s < samples[i].size(); s++) {
298  double denominator = 0.0;
299  for (int k = 0; k < n_windows; k++) {
300  denominator += window_cache[i][k][s] * z_inv[k];
301  }
302  denominator_cache[s] = 1 / denominator;
303  }
304 
305  for (int j = 0; j < n_windows; j++) {
306  double sum = 0.0;
307  int count = 0;
308 
309  for (size_t s = 0; s < samples[i].size(); s++) {
310  double sample = samples[i][s];
311  double window_j = window_cache[i][j][s];
312  double denominator = denominator_cache[s];
313 
314  if (denominator > 0) {
315  double integrand = (window_j * z_inv[i]) * denominator;
316  sum += integrand;
317  count++;
318  } else if (debug_mode) {
319  MACH3LOG_WARN("Denominator is zero for sample {} in window {}, skipping...", sample, i);
320  }
321  }
322 
323  if (debug_mode) {
324  MACH3LOG_INFO("F[{}][{}] sum: {}, count: {}", i, j, sum, count);
325  }
326 
327  if (count > 0) {
328  F[i][j] = sum / count;
329  }
330  }
331  }
332 
333  return F;
334 }

441 // This implements a check on stalling of the evolution of the window weights. Seems to be a better definition for convergence that above
442 // moving-average stalled-convergence check on z values
443 bool checkConvergenceStalled(const std::vector<double> &z_current, const std::vector<double> &z_prev, double tolerance) {
444  (void)z_prev;
445 
446  static std::deque<std::vector<double>> z_history;
447  static std::vector<double> previous_moving_average;
448  static int stagnant_iterations = 0;
449 
450  constexpr int moving_average_window = 500;
451  constexpr int stagnant_required = 500;
452  const double bound = tolerance;
453 
454  if (z_current.empty()) {
455  return false;
456  }
457 
458  // Reset state safely if number of windows changes between solver runs.
459  if (!z_history.empty() && z_history.front().size() != z_current.size()) {
460  z_history.clear();
461  previous_moving_average.clear();
462  stagnant_iterations = 0;
463  }
464 
465  z_history.push_back(z_current);
466  if (static_cast<int>(z_history.size()) > moving_average_window) {
467  z_history.pop_front();
468  }
469 
470  // Wait until the moving-average window is fully populated.
471  if (static_cast<int>(z_history.size()) < moving_average_window) {
472  return false;
473  }
474 
475  std::vector<double> moving_average(z_current.size(), 0.0);
476  for (const auto &z_vec : z_history) {
477  for (size_t i = 0; i < z_vec.size(); i++) {
478  moving_average[i] += z_vec[i];
479  }
480  }
481  for (size_t i = 0; i < moving_average.size(); i++) {
482  moving_average[i] /= moving_average_window;
483  }
484 
485  if (previous_moving_average.empty()) {
486  previous_moving_average = moving_average;
487  return false;
488  }
489 
490  bool all_within_bound = true;
491  for (size_t i = 0; i < moving_average.size(); i++) {
492  if (std::abs(moving_average[i] - previous_moving_average[i]) > bound) {
493  all_within_bound = false;
494  break;
495  }
496  }
497 
498  if (all_within_bound) {
499  stagnant_iterations++;
500  } else {
501  stagnant_iterations = 0;
502  }
503 
504  previous_moving_average = moving_average;
505 
506  if (stagnant_iterations == stagnant_required) {
507  MACH3LOG_WARN("Convergence appears stalled: moving-average change stayed within {} for {} iterations.",
508  bound, stagnant_required);
509  }
510 
511  return stagnant_iterations >= stagnant_required;
512 }
